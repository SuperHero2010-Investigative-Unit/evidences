## Mã nguồn C++ chuẩn

#include <iostream>#include <vector>#include <cmath>#include <algorithm>
using namespace std;
int main() {
    // Tối ưu hóa tốc độ nhập xuất dữ liệu
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n;
    if (!(cin >> n)) return 0;

    vector<double> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }

    // Bước 1: Sắp xếp dãy số tăng dần
    sort(a.begin(), a.end());

    // Bước 2: Đếm tần suất xuất hiện của các giá trị phân biệt
    vector<int> frequencies;
    int count = 1;
    
    for (int i = 1; i < n; ++i) {
        // Vì là số thực, có thể dùng trị tuyệt đối để so sánh bằng (sai số rất nhỏ)
        // Tuy nhiên với bài toán điểm số trắc nghiệm bước nhảy 0.01, so sánh trực tiếp == vẫn an toàn
        if (abs(a[i] - a[i - 1]) < 1e-9) {
            count++;
        } else {
            frequencies.push_back(count);
            count = 1;
        }
    }
    frequencies.push_back(count); // Thêm tần suất của phần tử cuối cùng

    // Bước 3: Tính tổng độ lệch tần số P
    long long P = 0; // Dùng long long đề phòng tổng lớn
    if (frequencies.size() > 1) {
        for (size_t i = 0; i < frequencies.size() - 1; ++i) {
            P += abs(frequencies[i] - frequencies[i + 1]);
        }
    }

    // In kết quả
    cout << P << "\n";

    return 0;
}

## Mã nguồn C++ (Kỹ thuật Đếm phân phối)

#include <iostream>#include <vector>#include <cmath>
using namespace std;
// Mảng đếm tần suất cho 1001 mức điểm (từ 0 đến 1000)int chuong[1005]; 
int main() {
    // Tối ưu tốc độ nhập xuất
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n;
    if (!(cin >> n)) return 0;

    // Bước 1: Lùa bò vào chuồng
    for (int i = 0; i < n; ++i) {
        double diem;
        cin >> diem;
        // Quy đổi số thực sang số nguyên từ 0 đến 1000 để làm index
        int index = round(diem * 100); 
        chuong[index]++;
    }

    // Bước 2: Gom các tần suất của các điểm có xuất hiện (vùng có bò)
    vector<int> tan_suat;
    for (int i = 0; i <= 1000; ++i) {
        if (chuong[i] > 0) {
            tan_suat.push_back(chuong[i]);
        }
    }

    // Bước 3: Tính tổng độ phân hóa P giữa các chuồng liên tiếp có bò
    long long P = 0;
    if (tan_suat.size() > 1) {
        for (size_t i = 0; i < tan_suat.size() - 1; ++i) {
            P += abs(tan_suat[i] - tan_suat[i + 1]);
        }
    }

    cout << P << "\n";

    return 0;
}

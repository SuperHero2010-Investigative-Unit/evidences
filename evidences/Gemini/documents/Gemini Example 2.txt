def doan_con_1(n, a):
    max_len = 1
    current_len = 1
    for i in range(1, n):
        if a[i] == a[i-1]:
            current_len += 1
        else:
            current_len = 1
        max_len = max(max_len, current_len)
    return max_len
def doan_con_2(n, a):
    if n == 0: return 0
    max_len = 1
    current_len = 1
    for i in range(1, n):
        # Kiểm tra khác dấu
        if (a[i] > 0 and a[i-1] < 0) or (a[i] < 0 and a[i-1] > 0):
            current_len += 1
        else:
            current_len = 1
        max_len = max(max_len, current_len)
    return max_len
def doan_con_3(n, a):
    max_so_far = float('-inf')
    max_ending_here = 0
    for x in a:
        max_ending_here += x
        if max_so_far < max_ending_here:
            max_so_far = max_ending_here
        if max_ending_here < 0:
            max_ending_here = 0
    return max_so_far

def count_adjacent_sum(arr, x):
    count = 0
    for i in range(len(arr) - 1):
        if arr[i] + arr[i+1] == x:
            count += 1
    return count
def count_any_pair_sum(arr, x):
    count = 0
    n = len(arr)
    for i in range(n):
        for j in range(i + 1, n):
            if arr[i] + arr[j] == x:
                count += 1
    return count
def is_prime(n):
    if n < 2: return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

def count_prime_sum_pairs(arr):
    count = 0
    n = len(arr)
    for i in range(n):
        for j in range(i + 1, n):
            if is_prime(arr[i] + arr[j]):
                count += 1
    return count

import math

def is_prime(n):
    if n < 2:
        return False

    for i in range(2, math.isqrt(n) + 1):
        if n % i == 0:
            return False

    return True

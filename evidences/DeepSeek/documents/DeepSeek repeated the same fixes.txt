## For truly streaming operations

If you want to avoid memory spikes, you need to use the standard library's `sys.io.File.update()` with chunked reading and writing:

```haxe
function insertIntoFileStreaming(filePath:String, insertPosition:Int, contentToInsert:String):Void
{
    var source = sys.io.File.update(filePath, false);
    var target = sys.io.File.write(filePath + ".tmp");
    
    // Copy first part up to insert position
    source.seek(0, SeekBegin);
    var chunkSize = 8192;
    var remaining = insertPosition;
    while (remaining > 0) {
        var chunk = source.read(Math.min(chunkSize, remaining));
        target.write(chunk);
        remaining -= chunk.length;
    }
    
    // Write the new content
    target.writeString(contentToInsert);
    
    // Copy the rest
    while (true) {
        var chunk = source.read(chunkSize);
        if (chunk.length == 0) break;
        target.write(chunk);
    }
    
    source.close();
    target.close();
    
    // Replace original with temp
    sys.io.File.delete(filePath);
    sys.io.File.rename(filePath + ".tmp", filePath);
}
```

This still uses a temp file but keeps memory usage low because it reads and writes in chunks. This is the most memory-efficient solution without external libraries.

private function appendTxtToTempFile(tempPath:String, nextTxtContent:String):Void
{
    // Read existing file
    var content = sys.io.File.getContent(tempPath);
    
    // Parse both files into structures
    var baseSections = parseTxtSections(content);
    var nextSections = parseTxtSections(nextTxtContent);
    
    // Merge sections by index
    var mergedSections = [];
    var maxSections = Std.int(Math.max(baseSections.length, nextSections.length));
    
    for (i in 0...maxSections) {
        var baseSection = i < baseSections.length ? baseSections[i] : createEmptySection();
        var nextSection = i < nextSections.length ? nextSections[i] : createEmptySection();
        
        // Merge sectionNotes
        var mergedNotes = [];
        if (baseSection.sectionNotes != null) mergedNotes = mergedNotes.concat(baseSection.sectionNotes);
        if (nextSection.sectionNotes != null) mergedNotes = mergedNotes.concat(nextSection.sectionNotes);
        
        baseSection.sectionNotes = mergedNotes;
        mergedSections.push(baseSection);
    }
    
    // Rebuild the TXT content with merged sections
    var mergedContent = rebuildTxtContent(content, mergedSections);
    
    // Write back
    sys.io.File.saveContent(tempPath, mergedContent);
}

private function parseTxtSections(txtContent:String):Array<Dynamic>
{
    var sections = [];
    var notesStart = txtContent.indexOf('notes: [\n');
    if (notesStart == -1) return sections;
    
    var notesEnd = txtContent.indexOf('\n]\n', notesStart);
    if (notesEnd == -1) return sections;
    
    var notesContent = txtContent.substring(notesStart + 9, notesEnd);
    var sectionPattern = ~/\[\n([\s\S]*?)\n    \]/g;
    var start = 0;
    
    while (sectionPattern.matchSub(notesContent, start)) {
        var sectionText = sectionPattern.matched(1);
        var section = parseTxtSection(sectionText);
        sections.push(section);
        start = sectionPattern.matchedPos().pos + sectionPattern.matchedPos().len;
    }
    
    return sections;
}

private function parseTxtSection(sectionText:String):Dynamic
{
    var section:Dynamic = {};
    var lines = sectionText.split('\n');
    for (line in lines) {
        line = StringTools.trim(line);
        if (line.indexOf(':') == -1) continue;
        var parts = line.split(':');
        if (parts.length < 2) continue;
        var key = StringTools.trim(parts[0]);
        var value = StringTools.trim(parts[1]);
        if (value.endsWith(',')) value = value.substring(0, value.length - 1);
        
        if (key == 'sectionNotes') {
            // Parse the array
            try {
                var notesArray = haxe.Json.parse(value);
                Reflect.setField(section, key, notesArray);
            } catch(e:Dynamic) {
                Reflect.setField(section, key, []);
            }
        } else if (value == 'true') {
            Reflect.setField(section, key, true);
        } else if (value == 'false') {
            Reflect.setField(section, key, false);
        } else if (value.startsWith('"')) {
            Reflect.setField(section, key, value.substring(1, value.length - 1));
        } else {
            Reflect.setField(section, key, Std.parseFloat(value));
        }
    }
    return section;
}

private function createEmptySection():Dynamic

    return {
        gfSection: false,
        altAnim: false,
        sectionNotes: [],
        bpm: 150,
        sectionBeats: 4,
        changeBPM: false,
        mustHitSection: true
    };

private function rebuildTxtContent(originalContent:String, sections:Array<Dynamic>):String
{
    // Build the notes array with merged sections
    var notesContent = "notes: [\n";
    for (i in 0...sections.length) {
        var section = sections[i];
        notesContent += '    [\n';
        notesContent += '        gfSection: ' + (section.gfSection ? 'true' : 'false') + ',\n';
        notesContent += '        altAnim: ' + (section.altAnim ? 'true' : 'false') + ',\n';
        notesContent += '        sectionNotes: ' + haxe.Json.stringify(section.sectionNotes) + ',\n';
        notesContent += '        bpm: ' + section.bpm + ',\n';
        notesContent += '        sectionBeats: ' + section.sectionBeats + ',\n';
        notesContent += '        changeBPM: ' + (section.changeBPM ? 'true' : 'false') + ',\n';
        notesContent += '        mustHitSection: ' + (section.mustHitSection ? 'true' : 'false') + '\n';
        notesContent += '    ]';
        if (i < sections.length - 1) notesContent += ',\n';
        else notesContent += '\n';
    }

private function findArrayEnd(file:sys.io.FileInput, arrayName:String, fileSize:Int):Int
{
    var chunkSize = 8192;
    var buffer = "";
    var pos = 0;
    var foundStart = false;
    var bracketCount = 0;
    var inString = false;
    var escapeNext = false;
    var arrayEnd = -1;
    
    file.seek(0, SeekBegin);
    
    while (pos < fileSize) {
        var chunk = file.read(chunkSize).toString();
        buffer += chunk;
        pos += chunk.length;
        
        if (!foundStart) {
            var startPos = buffer.indexOf('"' + arrayName + '"');
            if (startPos != -1) {
                for (i in startPos...buffer.length) {
                    var char = buffer.charAt(i);
                    if (escapeNext) {
                        escapeNext = false;
                        continue;
                    }
                    if (char == '\\') {
                        escapeNext = true;
                        continue;
                    }
                    if (char == '"') {
                        inString = !inString;
                        continue;
                    }
                    if (!inString && char == '[') {
                        foundStart = true;
                        bracketCount = 1;
                        break;
                    }
                }
            }
        }
        
        if (foundStart) {
            var startIdx = 0;
            if (buffer.length > chunkSize * 2) {
                startIdx = buffer.length - chunkSize * 2;
            }
            for (i in startIdx...buffer.length) {
                var char = buffer.charAt(i);
                if (escapeNext) {
                    escapeNext = false;
                    continue;
                }
                if (char == '\\') {
                    escapeNext = true;
                    continue;
                }
                if (char == '"') {
                    inString = !inString;
                    continue;
                }
                if (!inString) {
                    if (char == '[') bracketCount++;
                    else if (char == ']') {
                        bracketCount--;
                        if (bracketCount == 0) {
                            arrayEnd = (pos - buffer.length) + i;
                            return arrayEnd;
                        }
                    }
                }
            }
        }
        
        if (buffer.length > chunkSize * 10) {
            buffer = buffer.substr(-chunkSize * 5);
        }
    }
    
    return -1;
}

private function copyChunk(source:sys.io.FileInput, target:sys.io.FileOutput, bytes:Int):Void
{
    var chunkSize = 8192;
    var remaining = bytes;
    while (remaining > 0) {
        var toRead = Math.min(chunkSize, remaining);
        var chunk = source.read(toRead);
        if (chunk.length == 0) break;
        target.write(chunk);
        remaining -= toRead;
    }
}

private function replaceFile(original:String, temp:String):Void
{
    if (FileSystem.exists(original)) FileSystem.deleteFile(original);
    FileSystem.rename(temp, original);
}

5. **Memory usage** is limited to the chunk size (8KB)

private function findArrayEndInPlace(file:sys.io.FileInput, arrayName:String, fileSize:Int):Int
{
    var chunkSize = 8192;
    var buffer = "";
    var pos = 0;
    var foundStart = false;
    var bracketCount = 0;
    var inString = false;
    var escapeNext = false;
    var arrayEnd = -1;
    
    file.seek(0, SeekBegin);
    
    while (pos < fileSize) {
        var chunk = file.read(chunkSize).toString();
        buffer += chunk;
        pos += chunk.length;
        
        if (!foundStart) {
            var startPos = buffer.indexOf('"' + arrayName + '"');
            if (startPos != -1) {
                for (i in startPos...buffer.length) {
                    var char = buffer.charAt(i);
                    if (escapeNext) {
                        escapeNext = false;
                        continue;
                    }
                    if (char == '\\') {
                        escapeNext = true;
                        continue;
                    }
                    if (char == '"') {
                        inString = !inString;
                        continue;
                    }
                    if (!inString && char == '[') {
                        foundStart = true;
                        bracketCount = 1;
                        break;
                    }
                }
            }
        }
        
        if (foundStart) {
            var startIdx = 0;
            if (buffer.length > chunkSize * 2) {
                startIdx = buffer.length - chunkSize * 2;
            }
            for (i in startIdx...buffer.length) {
                var char = buffer.charAt(i);
                if (escapeNext) {
                    escapeNext = false;
                    continue;
                }
                if (char == '\\') {
                    escapeNext = true;
                    continue;
                }
                if (char == '"') {
                    inString = !inString;
                    continue;
                }
                if (!inString) {
                    if (char == '[') bracketCount++;
                    else if (char == ']') {
                        bracketCount--;
                        if (bracketCount == 0) {
                            arrayEnd = (pos - buffer.length) + i;
                            return arrayEnd;
                        }
                    }
                }
            }
        }
        
        if (buffer.length > chunkSize * 10) {
            buffer = buffer.substr(-chunkSize * 5);
        }
    }
    
    return -1;
}

1. `file.seek(position, SeekBegin)` - move to the insertion point

import openfl.filesystem.FileStream;
import openfl.filesystem.File;
import openfl.filesystem.FileMode;
import openfl.events.Event;

private function appendTxtToTempFile(tempPath:String, nextTxtContent:String):Void
{
    trace('=== appendTxtToTempFile START ===');
    trace('tempPath: ' + tempPath);
    
    // Extract notes and events from the new TXT content
    var notesStart = nextTxtContent.indexOf('notes: [');
    var notesEnd = nextTxtContent.indexOf('\n]', notesStart);
    var eventsStart = nextTxtContent.indexOf('events: [');
    var eventsEnd = nextTxtContent.indexOf('\n]', eventsStart);
    
    var newNotesContent = "";
    var newEventsContent = "";
    
    if (notesStart != -1 && notesEnd != -1) {
        newNotesContent = nextTxtContent.substring(notesStart + 9, notesEnd);
    }
    if (eventsStart != -1 && eventsEnd != -1) {
        newEventsContent = nextTxtContent.substring(eventsStart + 9, eventsEnd);
    }
    
    if (newNotesContent.length == 0 && newEventsContent.length == 0) {
        trace('No new content');
        return;
    }
    
    // Use FileStream with UPDATE mode for in-place modification
    var file = new FileStream();
    file.addEventListener(Event.COMPLETE, function(e) {
        trace('File operation complete');
    });
    file.openAsync(File.applicationDirectory.resolvePath(tempPath), FileMode.UPDATE);
    
    // Find notes array end position
    var notesArrayEnd = findArrayEndWithFileStream(file, "notes");
    if (notesArrayEnd == -1) {
        trace('ERROR: Could not find notes array end');
        file.close();
        return;
    }
    
    // Insert notes content
    if (newNotesContent.length > 0) {
        file.position = notesArrayEnd;
        var rest = file.readUTFBytes(file.bytesAvailable);
        file.position = notesArrayEnd;
        file.writeUTFBytes(",\n" + newNotesContent);
        file.writeUTFBytes(rest);
    }
    
    // Find events array end position
    var eventsArrayEnd = findArrayEndWithFileStream(file, "events");
    if (eventsArrayEnd != -1 && newEventsContent.length > 0) {
        file.position = eventsArrayEnd;
        var rest = file.readUTFBytes(file.bytesAvailable);
        file.position = eventsArrayEnd;
        file.writeUTFBytes(",\n" + newEventsContent);
        file.writeUTFBytes(rest);
    }
    
    file.close();
    trace('=== appendTxtToTempFile COMPLETE ===');
}

private function findArrayEndWithFileStream(file:FileStream, arrayName:String):Int
{
    var chunkSize = 8192;
    var buffer = "";
    var pos = 0;
    var foundStart = false;
    var bracketCount = 0;
    var inString = false;
    var escapeNext = false;
    var arrayEnd = -1;
    
    file.position = 0;
    var fileSize = file.bytesAvailable;
    
    while (pos < fileSize) {
        var chunk = file.readUTFBytes(Math.min(chunkSize, fileSize - pos));
        buffer += chunk;
        pos += chunk.length;
        
        if (!foundStart) {
            var startPos = buffer.indexOf('"' + arrayName + '"');
            if (startPos != -1) {
                for (i in startPos...buffer.length) {
                    var char = buffer.charAt(i);
                    if (escapeNext) { escapeNext = false; continue; }
                    if (char == '\\') { escapeNext = true; continue; }
                    if (char == '"') { inString = !inString; continue; }
                    if (!inString && char == '[') {
                        foundStart = true;
                        bracketCount = 1;
                        break;
                    }
                }
            }
        }
        
        if (foundStart) {
            for (i in 0...buffer.length) {
                var char = buffer.charAt(i);
                if (escapeNext) { escapeNext = false; continue; }
                if (char == '\\') { escapeNext = true; continue; }
                if (char == '"') { inString = !inString; continue; }
                if (!inString) {
                    if (char == '[') bracketCount++;
                    else if (char == ']') {
                        bracketCount--;
                        if (bracketCount == 0) {
                            arrayEnd = (pos - buffer.length) + i;
                            return arrayEnd;
                        }
                    }
                }
            }
        }
        
        if (buffer.length > chunkSize * 10) {
            buffer = buffer.substr(-chunkSize * 5);
        }
    }
    
    return -1;
}

This uses `FileStream` with `FileMode.UPDATE` which allows:
1. **Append** – writing at the end
2. **Insert in middle** – by reading the rest of the file and writing it back after the insertion
3. **Async** – via `openAsync()` with `Event.COMPLETE` listener

The `FileStream` also supports `readUTFBytes` and `writeUTFBytes` which are simpler than the problematic `sys.io.File` functions.
